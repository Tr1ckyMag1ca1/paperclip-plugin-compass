```typescript
import React, { useState } from "react";
import { usePluginAction, usePluginData } from "@paperclipai/plugin-sdk/ui";
import { classifyStall } from "../../revive/classify.js";
import { ActionQueuePanel } from "./ActionQueuePanel.js";
import { StallSummaryBadge } from "./StallSummaryBadge.js";
import { useReviveRunState } from "./ReviveRunState.js";

export const RevivePanel: React.FC = () => {
  const [queue, setQueue] = useReviveRunState();
  const [isLoading, setIsLoading] = useState(false);
  
  const { inventory } = usePluginData(); // InventorySnapshot from Phase 1
  const { classifyStall: classifyStallAction } = usePluginAction();

  const handleDiagnose = async () => {
    setIsLoading(true);
    try {
      // Call worker handler to run classifier
      const result = await classifyStallAction({
        snapshot: inventory,
        // ... other params
      });
      setQueue(result.queue);
    } catch (error) {
      // Error display
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="flex flex-col h-full bg-background">
      {/* Fixed Header */}
      <header className="p-lg border-b border-border">
        <h1 className="text-display font-bold">{inventory?.companyName}</h1>
        <p className="text-body text-foreground/70">Fix what's blocking this company</p>
        
        {inventory && <StallSummaryBadge inventory={inventory} />}
        
        <button
          onClick={handleDiagnose}
          disabled={isLoading}
          className="mt-md px-lg py-md bg-accent text-white rounded"
        >
          {isLoading ? "Diagnosing…" : "Find what's blocking this company"}
        </button>
      </header>

      {/* Main Content */}
      <main className="flex-1 overflow-y-auto p-lg">
        {isLoading ? (
          <div>Analyzing blockers…</div>
        ) : !queue || queue.items.length === 0 ? (
          <EmptyReviveState />
        ) : (
          <ActionQueuePanel queue={queue} />
        )}
      </main>

      {/* Sticky Footer */}
      {queue && queue.total_items > 0 && (
        <footer className="sticky bottom-0 border-t border-border bg-card p-lg flex justify-between items-center">
          <span className="text-label text-foreground/70">
            {queue.addressed_count} of {queue.total_items} addressed
          </span>
          <button
            disabled={queue.addressed_count === 0}
            className="px-lg py-md bg-accent text-white rounded"
          >
            Review and apply
          </button>
        </footer>
      )}
    </div>
  );
};

const EmptyReviveState: React.FC = () => (
  <div className="text-center py-3xl">
    <h2 className="text-heading font-bold mb-md">This company isn't stalled</h2>
    <p className="text-body mb-lg">
      No blocking issues detected. Try Assess for a strategic audit instead.
    </p>
  </div>
);
```

```typescript
import React from "react";
import type { ActionQueue, StallCause } from "../../types/revive.js";
import { ActionItemCard } from "./ActionItemCard.js";

const CAUSE_HEADERS: Record<StallCause, string> = {
  "single-blocker": "Stuck on a blocker",
  "strategic-drift": "Drifted from vision",
  "broken-integration": "Integration broken",
  "governance-loop": "Stuck in approval loop",
  "dead-agent": "Agent stopped responding",
};

interface ActionQueuePanelProps {
  queue: ActionQueue;
  onActionApply?: (actionId: string) => Promise<void>;
}

export const ActionQueuePanel: React.FC<ActionQueuePanelProps> = ({ queue, onActionApply }) => {
  return (
    <div className="space-y-lg">
      {Object.entries(queue.items_by_cause).map(([cause, items]) => {
        if (!items || items.length === 0) return null;
        
        return (
          <section key={cause}>
            <h3 className="text-heading font-bold mb-md">{CAUSE_HEADERS[cause as StallCause]}</h3>
            <div className="space-y-sm">
              {items
                .sort((a, b) => b.priority - a.priority) // High priority first
                .map((item) => (
                  <ActionItemCard
                    key={item.id}
                    item={item}
                    onApply={() => onActionApply?.(item.id)}
                  />
                ))}
            </div>
          </section>
        );
      })}
    </div>
  );
};
```

```typescript
import React, { useState } from "react";
import type { ActionItem } from "../../types/revive.js";
import { PriorityBadge } from "./PriorityBadge.js";
import { SamplePivotModal } from "./SamplePivotModal.js";
import { ActionConfirmationModal } from "./ActionConfirmationModal.js";

interface ActionItemCardProps {
  item: ActionItem;
  onApply?: () => Promise<void>;
}

export const ActionItemCard: React.FC<ActionItemCardProps> = ({ item, onApply }) => {
  const [showExplain, setShowExplain] = useState(false);
  const [showConfirm, setShowConfirm] = useState(false);
  const [isApplying, setIsApplying] = useState(false);

  const handleApply = async () => {
    if (item.recommended_action.type === "pivot-to-sample") {
      // Sample-pivot has its own modal
      setShowConfirm(true);
    } else {
      // Standard action confirmation
      setShowConfirm(true);
    }
  };

  const handleConfirm = async () => {
    setIsApplying(true);
    try {
      await onApply?.();
      setShowConfirm(false);
    } finally {
      setIsApplying(false);
    }
  };

  return (
    <>
      <div className="p-md bg-card rounded border border-border">
        {/* Header: Priority | Title | Status */}
        <div className="flex items-start gap-md mb-md">
          <PriorityBadge priority={item.priority} />
          <div className="flex-1">
            <h4 className="text-body font-bold">{item.title}</h4>
            <p className="text-label text-foreground/70 mt-xs">
              Unlocks {item.unblocks_count || 1} downstream issue(s)
            </p>
          </div>
          <span className="text-label text-foreground/70">
            {item.status === "pending" ? "Pending" : item.status === "addressed" ? "✓ Addressed" : "× Dismissed"}
          </span>
        </div>

        {/* Body: Description */}
        <p className="text-body mb-md">{item.why_blocking}</p>

        {/* Action Buttons */}
        <div className="flex gap-md">
          <button
            onClick={handleApply}
            disabled={item.status !== "pending" || isApplying}
            className="flex-1 px-md py-sm bg-accent text-white rounded"
          >
            {isApplying ? "Applying…" : getActionButtonLabel(item.recommended_action.type)}
          </button>
          <button
            onClick={() => setShowExplain(!showExplain)}
            className="px-md py-sm text-foreground border border-border rounded"
          >
            Explain
          </button>
          <button className="px-md py-sm text-destructive border border-border rounded">
            Dismiss
          </button>
        </div>

        {/* Collapsible Explanation */}
        {showExplain && (
          <div className="mt-md p-md bg-background rounded text-body">
            {item.title}: {item.why_blocking}
          </div>
        )}
      </div>

      {/* Confirmation Modal */}
      {showConfirm && item.recommended_action.type === "pivot-to-sample" ? (
        <SamplePivotModal
          onConfirm={handleConfirm}
          onCancel={() => setShowConfirm(false)}
        />
      ) : (
        <ActionConfirmationModal
          action={item}
          onConfirm={handleConfirm}
          onCancel={() => setShowConfirm(false)}
        />
      )}
    </>
  );
};

function getActionButtonLabel(type: string): string {
  const labels: Record<string, string> = {
    "replace-blocker-issue": "Replace this issue",
    "reassign-issue": "Reassign to {agent}",
    "nudge-agent-with-context-doc": "Brief the agent",
    "pivot-to-sample": "Switch to sample mode",
    "mark-blocker-resolved": "Mark resolved",
    "restart-agent": "Restart agent",
    "surface-amendment-needed": "Review in Assess",
  };
  return labels[type] || "Apply action";
}
```

```typescript
import { useState, useEffect } from "react";
import { usePluginState } from "@paperclipai/plugin-sdk/ui";
import type { ActionQueue } from "../../types/revive.js";

/**
 * Hook to persist Revive diagnosis in worker-state across panel reloads.
 * Per D-15: stored under key `compass:revive:run:${company_id}`.
 */
export function useReviveRunState(): [ActionQueue | null, (queue: ActionQueue | null) => void] {
  const { state, setState } = usePluginState();
  const [queue, setQueueLocal] = useState<ActionQueue | null>(null);

  const stateKey = "compass:revive:run:current";

  // Load from state on mount
  useEffect(() => {
    const cached = state?.[stateKey] as ActionQueue | undefined;
    if (cached) {
      setQueueLocal(cached);
    }
  }, [state, stateKey]);

  const setQueue = (newQueue: ActionQueue | null) => {
    setQueueLocal(newQueue);
    if (newQueue) {
      setState({ ...state, [stateKey]: newQueue });
    } else {
      const updated = { ...state };
      delete updated[stateKey];
      setState(updated);
    }
  };

  return [queue, setQueue];
}
```
