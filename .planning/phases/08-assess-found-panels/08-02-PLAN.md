   ```typescript
   const CONFIDENCE_FILL_CLASSES = {
     high: "bg-emerald-500",     // confidence > 0.75
     medium: "bg-yellow-500",    // 0.4 <= confidence <= 0.75
     low: "bg-red-500"           // confidence < 0.4
   };
   
   function getConfidenceLevel(confidence: number): "high" | "medium" | "low" {
     if (confidence > 0.75) return "high";
     if (confidence >= 0.4) return "medium";
     return "low";
   }
   ```

3. **Component implementation pattern:**
   ```typescript
   export function ConfidenceBar({ confidence, label = true }: ConfidenceBarProps): React.ReactElement {
     const level = getConfidenceLevel(confidence);
     const fillPercentage = Math.round(confidence * 100);
   
     return (
       <div className="flex items-center gap-2">
         {/* Track + fill */}
         <div className="flex-1 h-2 bg-muted rounded-none overflow-hidden">
           <div
             className={`h-full ${CONFIDENCE_FILL_CLASSES[level]}`}
             style={{ width: `${fillPercentage}%` }}
           />
         </div>
         {/* Label */}
         {label && (
           <span className="text-xs font-medium text-muted-foreground whitespace-nowrap">
             {fillPercentage}%
           </span>
         )}
       </div>
     );
   }
   ```
