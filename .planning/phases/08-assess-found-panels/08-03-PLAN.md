   ```typescript
   export function AmendmentDiff({
     amendment,
     className = "",
   }: AmendmentDiffProps): React.ReactElement {
     const [isOpen, setIsOpen] = useState(false);
     const lines = amendment.split("\n").filter(line => line.trim().length > 0);

     return (
       <details
         open={isOpen}
         onToggle={e => setIsOpen(e.currentTarget.open)}
         className={`group ${className}`}
       >
         {/* Summary: collapsible trigger */}
         <summary className="cursor-pointer flex items-center gap-2 text-xs font-medium text-foreground hover:text-foreground/80 transition-colors p-2 hover:bg-muted rounded-none select-none">
           <ChevronDown className="h-4 w-4 transition-transform group-open:rotate-180" />
           <span>Proposed amendment:</span>
         </summary>

         {/* Diff content with semantic highlighting */}
         <div className="mt-3 p-3 bg-muted rounded-none border border-border overflow-x-auto">
           <pre className="text-xs font-normal leading-relaxed whitespace-pre-wrap break-words">
             {lines.map((line, idx) => {
               if (line.startsWith("+")) {
                 return (
                   <div key={idx} className="text-emerald-600 bg-emerald-500/10">
                     {line}
                   </div>
                 );
               } else if (line.startsWith("-")) {
                 return (
                   <div key={idx} className="text-red-600 bg-red-500/10">
                     {line}
                   </div>
                 );
               } else {
                 return (
                   <div key={idx} className="text-foreground">
                     {line}
                   </div>
                 );
               }
             })}
           </pre>
         </div>
       </details>
     );
   }
   ```

   import React from "react";
   import { describe, it, expect } from "vitest";
   import { render, screen } from "@testing-library/react";
   import userEvent from "@testing-library/user-event";
   import { AmendmentDiff } from "../src/ui/assess/AmendmentDiff";

   describe("AmendmentDiff line highlighting", () => {
     const sampleDiff = `+const x = 1
−const x = 0
 const y = 2`;

     it("renders add lines with emerald color", () => {
       const { container } = render(<AmendmentDiff amendment={sampleDiff} />);
       expect(container.innerHTML).toContain("text-emerald-600");
       expect(container.innerHTML).toContain("+const x = 1");
     });

     it("renders remove lines with red color", () => {
       const { container } = render(<AmendmentDiff amendment={sampleDiff} />);
       expect(container.innerHTML).toContain("text-red-600");
       expect(container.innerHTML).toContain("−const x = 0");
     });

     it("renders context lines neutral (no color background)", () => {
       const { container } = render(<AmendmentDiff amendment={sampleDiff} />);
       const contextDiv = Array.from(container.querySelectorAll("div")).find(
         el => el.textContent?.includes("const y = 2")
       );
       expect(contextDiv?.className).toContain("text-foreground");
       expect(contextDiv?.className).not.toContain("bg-emerald");
       expect(contextDiv?.className).not.toContain("bg-red");
     });

     it("details element can be toggled open/closed", async () => {
       const { container } = render(<AmendmentDiff amendment={sampleDiff} />);
       const details = container.querySelector("details");
       expect(details?.open).toBe(false);
       
       const summary = container.querySelector("summary");
       if (summary) await userEvent.click(summary);
       expect(details?.open).toBe(true);
     });

     it("preserves details/summary structure", () => {
       const { container } = render(<AmendmentDiff amendment={sampleDiff} />);
       expect(container.querySelector("details")).toBeTruthy();
       expect(container.querySelector("summary")).toBeTruthy();
     });

     it("contains no broken tokens", () => {
       const { container } = render(<AmendmentDiff amendment={sampleDiff} />);
       expect(container.innerHTML).not.toContain("gap-sm");
       expect(container.innerHTML).not.toContain("p-md");
       expect(container.innerHTML).not.toContain("rounded-lg");
       expect(container.innerHTML).not.toContain("text-accent");
     });
   });
