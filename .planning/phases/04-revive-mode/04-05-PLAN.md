import { classifyStall } from "./revive/classify.js";
import { applyAction, applyAllActions } from "./revive/apply.js";
import { writeActionQueueDocument } from "./revive/queue.js";
import type { StallClassification } from "./types/revive.js";

// Handler 1: runStallDiagnostic
plugin.addHandler("revive:runStallDiagnostic", async (ctx, params: { companyId: string }) => {
  try {
    const { snapshot, vision, activity } = await prepareReviveInputs(ctx, params.companyId);
    
    // Per D-02: deterministic classifier, no LLM
    const classification = classifyStall(snapshot, vision, activity);
    
    // Generate action queue from classification
    const actionQueue = generateActionQueueFromClassification(classification);
    
    // Write queue to documents (idempotent per D-04)
    await writeActionQueueDocument(ctx.adapter, params.companyId, actionQueue);
    
    return { success: true, queue: actionQueue };
  } catch (error) {
    return { success: false, error: String(error) };
  }
});

// Handler 2: applyReviveAction
plugin.addHandler(
  "revive:applyReviveAction",
  async (ctx, params: { companyId: string; actionId: string; queueRunId: string }) => {
    try {
      // Load queue from documents
      const queue = await loadActionQueue(ctx, params.companyId, params.queueRunId);
      if (!queue) {
        return { success: false, error: "Queue not found" };
      }
      
      // Apply single action (per D-10: incremental, not transactional)
      const { queue: updatedQueue, result } = await applyAction(
        queue,
        params.actionId,
        ctx.adapter
      );
      
      return { success: result.success, summary: result.summary, queue: updatedQueue };
    } catch (error) {
      return { success: false, error: String(error) };
    }
  }
);

// Handler 3: checkReviveActionStatus
plugin.addHandler(
  "revive:checkReviveActionStatus",
  async (ctx, params: { companyId: string; queueRunId: string }) => {
    try {
      // Load queue from documents
      const queue = await loadActionQueue(ctx, params.companyId, params.queueRunId);
      if (!queue) {
        return { found: false };
      }
      
      return {
        found: true,
        totalItems: queue.total_items,
        addressedCount: queue.addressed_count,
        pendingCount: queue.total_items - queue.addressed_count,
      };
    } catch (error) {
      return { found: false, error: String(error) };
    }
  }
);

// Helper: Load queue from documents (idempotent)
async function loadActionQueue(ctx: any, companyId: string, runId: string) {
  const docKey = `compass:revive:action-queue:${runId}`;
  return await ctx.adapter.getDocument(companyId, docKey);
}

// Helper: Generate action queue from classification
function generateActionQueueFromClassification(classification: StallClassification) {
  // Convert StallClassification → ActionQueue with all 7 action types
  // Priority computed from confidence + blast radius per D-05
  // Items grouped by cause per D-06
  return {
    run_id: generateRunId(),
    company_id: classification.companyId,
    created_at: new Date().toISOString(),
    items_by_cause: {}, // Populated from classification causes
    total_items: 0,
    addressed_count: 0,
  };
}
