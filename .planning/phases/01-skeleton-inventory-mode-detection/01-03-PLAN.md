```typescript
// Retrieve stored override for this company
const companyId = ctx.companyContext?.id; // or extract from plugin context

const finalMode = storedOverride || autoDetectedMode;

// Store in context for UI access
ctx.state.set({
  scopeKind: "company",
  scopeId: companyId,
  namespace: "compass-state",
  stateKey: "current-mode"
}, finalMode);
```

```typescript
ctx.onAction("setModeOverride", async (mode: Mode) => {

  return { success: true, newMode: mode };
});
```

```typescript
export async function updateModeOverride(ctx: PluginContext, companyId: string, mode: Mode): Promise<void> {
```
