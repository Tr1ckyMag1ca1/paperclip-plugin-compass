3. **Implementation (TypeScript with Node fs/path):**
   ```typescript
   import { execSync } from "child_process";
   import path from "path";

   const BROKEN_PATTERNS = [
     "gap-xs", "gap-sm", "gap-md",
     "px-xs", "px-sm", "px-md",
     "py-xs", "py-sm", "py-md",
     "mt-xs", "mt-sm", "pb-sm",
     "text-heading", "text-body", "text-label",
     "bg-green-50", "bg-red-50", "bg-slate-50",
     "text-green-700", "text-red-700", "text-slate-600", "text-slate-700",
     "border-green-200", "border-red-200", "border-slate-200", "border-slate-300",
     "rounded-lg", "rounded-xl", "rounded-2xl"
   ];

   export async function verifyBrokenPatterns(scanPath: string): Promise<{
     passed: boolean;
     totalHits: number;
     details: { pattern: string; count: number; files: string[] }[];
   }> {
     const results: { pattern: string; count: number; files: string[] }[] = [];
     let totalHits = 0;

     for (const pattern of BROKEN_PATTERNS) {
       // Escape regex special chars in pattern
       const escapedPattern = pattern.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
       
       // Grep command: search src/ui/, exclude test files and DualRenderProbe
       const grepCmd = `grep -r "${escapedPattern}" src/ui/ --include="*.tsx" | grep -v ".test.tsx" | grep -v "DualRenderProbe.tsx" | grep -v "^[^:]*:#" || true`;
       
       try {
         const output = execSync(grepCmd, { encoding: "utf-8" }).trim();
         
         if (output) {
           const lines = output.split("\n").filter(l => l);
           const files = [...new Set(lines.map(l => l.split(":")[0]))];
           
           results.push({
             pattern,
             count: lines.length,
             files
           });
           
           totalHits += lines.length;
         }
       } catch (e) {
         // grep returns exit code 1 when no matches; ignore
       }
     }

     return {
       passed: totalHits === 0,
       totalHits,
       details: results
     };
   }
   ```
